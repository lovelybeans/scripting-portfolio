This is the function that handles transformation.


function Transform(plr: Player): ()
	local mirac = plr:GetAttribute(`Miraculous`)

	if mirac > max_mir or mirac <= 0 then return end
	if plr:GetAttribute(`busy`) then return end

	local char = plr.Character
	local hum: Humanoid = char.Humanoid
	local animator: Animator = hum.Animator

	if not plr:GetAttribute(`Transformed`) then
		if not plr or typeof(plr) ~= `Instance` or not plr:IsA(`Player`) then warn(`Player argument is not a player type.`); return false end

		plr:SetAttribute(`busy`, true)

		hum.WalkSpeed = 0

		local anim = animator:LoadAnimation(script.ButterflyTransform)
		local anims = anim:Play()

		anim:GetMarkerReachedSignal(`Pants`):Connect(function(name)
			if clothes[mirac] ~= nil then
				local cloth = clothes[mirac]

				char.Pants.PantsTemplate = `rbxassetid://{cloth.Pants.Value}`
			end
		end)

		anim:GetMarkerReachedSignal(`Shirt`):Connect(function(name)
			if clothes[mirac] ~= nil then
				local cloth = clothes[mirac]

				char.Shirt.ShirtTemplate = `rbxassetid://{cloth.Shirt.Value}`
			end
		end)

		anim.Stopped:Wait()

		hum.WalkSpeed = 40
		hum.JumpHeight = 15

		plr:SetAttribute(`busy`, false)
		plr:SetAttribute(`Transformed`, true)

	elseif plr:GetAttribute(`Transformed`) then
		plr:SetAttribute(`Transformed`, false)

		workspace.Rig.Humanoid:ApplyDescriptionResetAsync(workspace.Rig.Humanoid:GetAppliedDescription())
		workspace.Rig.Humanoid:ApplyDescriptionAsync(game.Players:GetHumanoidDescriptionFromUserIdAsync(plr.UserId))
		local a = workspace.Rig.Humanoid:GetAppliedDescription()

		char.Shirt.ShirtTemplate = workspace.Rig.Shirt.ShirtTemplate
		char.Pants.PantsTemplate = workspace.Rig.Pants.PantsTemplate

		hum.WalkSpeed = 16
		hum.JumpHeight = 7.2
	end
end

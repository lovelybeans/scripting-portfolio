local Players = game:GetService("Players")
local CharacterRoots = {}
local Chars = {}
local AnimID = 76475539920508

Players.PlayerAdded:Connect(function(plr)
	plr.CharacterAdded:Connect(function(char)
		if not CharacterRoots[plr.UserId] then
			local RootPart = char:WaitForChild("HumanoidRootPart")
			CharacterRoots[plr.UserId] = RootPart
			Chars[plr.UserId] = char
			Instance.new("Attachment").Parent = RootPart
			Instance.new("Animator").Parent = char.Humanoid
		else
			CharacterRoots[plr.UserId] = char:WaitForChild("HumanoidRootPart")
			Chars[plr.UserId] = char
			Instance.new("Attachment").Parent = CharacterRoots[plr.UserId]
			Instance.new("Animator").Parent = char.Humanoid
		end
	end)
end)

Shop Animation

The shop animation that makes it appear.


local ShopPos = Vector3.new(1.45, 0.209, 21.275)
local ShopkeeperPos = Vector3.new(1.594, 3.856, 19.304)
local ID = 131058991473907
local Tween = game:GetService("TweenService")
local TweenAnimShop = (function()
	local cf = Instance.new("CFrameValue")
	cf.Name = "ShopTweenCF"
	cf.Value = workspace.Shop:GetPrimaryPartCFrame()
	cf.Parent = script -- prevent garbage collection
	cf.Changed:Connect(function(v)
		workspace.Shop:SetPrimaryPartCFrame(v)
	end)
	return Tween:Create(cf, TweenInfo.new(3), {Value = CFrame.new(ShopPos)})
end)()

local TweenAnimKeeper = (function()
	local cf = Instance.new("CFrameValue")
	cf.Name = "KeeperTweenCF"
	cf.Value = workspace.Shopkeeper:GetPrimaryPartCFrame()
	cf.Parent = script -- prevent garbage collection
	cf.Changed:Connect(function(v)
		workspace.Shopkeeper:SetPrimaryPartCFrame(v)
	end)
	return Tween:Create(cf, TweenInfo.new(3), {Value = CFrame.new(ShopkeeperPos)})
end)()








script.Parent.Triggered:Connect(function()
	script.Parent.Enabled = false
	script.Parent.Parent.Humanoid.Animator:LoadAnimation(script.Animation):Play()
	task.wait(1.16)
	TweenAnimShop:Play()
	TweenAnimKeeper:Play()
	task.wait(2.75)
	workspace.AnimKeeper:Destroy()
	for i, v in pairs(workspace.Shopkeeper:GetChildren()) do
		if v:IsA("BasePart") then
			v.Transparency = 0
			if v.Name == "Head" then
				v.face.Transparency = 0
			end
		end
	end
	workspace.Shopkeeper.ProximityPrompt.Enabled = true
end)

Opening

This script is responsible for opening the door :D


local Players = game:GetService("Players")
local LED = script.Parent.Parent.Parent.CardScanner.LED
local CardPos = script.Parent.Parent.Parent.CardScanner.Part.Position
local CardRot = script.Parent.Parent.Parent.CardScanner.Part.Orientation
local Red = Color3.fromRGB(196, 40, 28)
local Yellow = Color3.fromRGB(255, 238, 0)
local Green = Color3.fromRGB(34, 255, 0)
local Tween = TweenInfo.new(
	1,
	Enum.EasingStyle.Quad
)
local Pos = {Position = Vector3.new(CardPos.X-0.123, CardPos.Y-1.226, CardPos.Z-0.192)}
local PosS = {Position = Vector3.new(CardPos.X-0.123, CardPos.Y-1.216, CardPos.Z-0.039)}
local KeyRank = 5

script.Parent.ObjectText = ("Minimum Level: " ..KeyRank)

script.Parent.Triggered:Connect(function(plr)
	local char = plr.Character
	local Keycard = plr.Backpack:FindFirstChild("Keycard") or char:FindFirstChild("Keycard")
	local KeyChar = char:FindFirstChild("Keycard")
	if Keycard then
		script.Parent.Enabled = false
		local KeyModel = Keycard.Handle:Clone()
		KeyModel.Anchored = true
		KeyModel.Position = Vector3.new(CardPos.X-0.123, CardPos.Y+1.224, CardPos.Z-0.192)
		KeyModel.Orientation = Vector3.new(CardRot.X, CardRot.Y+90, CardRot.Z+90)
		KeyModel.Parent = game.Workspace.Door.CardScanner
		if KeyChar then
			local Key = KeyChar:Clone()
			KeyChar:Destroy()
			Key.Parent = plr.Backpack
		end
		local TweenAn = game.TweenService:Create(KeyModel, Tween, Pos)
		local TweenAnS = game.TweenService:Create(KeyModel.Part, Tween, PosS)
		local NewKey = plr.Backpack:FindFirstChild("Keycard") or char:FindFirstChild("Keycard")
		task.wait(0.5)
		TweenAnS:Play()
		TweenAn:Play()
		task.wait(1)
		if NewKey.KeyCardID.Value >= KeyRank then
			LED.Color = Green
			script.Parent.DoorOpen:Fire()
			task.wait(0.5)
			KeyModel:Destroy()
			task.wait(1)
			LED.Color = Yellow

		else
			print("Error: Access Denied.")
			LED.Color = Red
			task.wait(0.2)
			LED.Color = Yellow
			task.wait(0.2)
			LED.Color = Red
			task.wait(0.2)
			LED.Color = Yellow
			task.wait(0.2)
			LED.Color = Red
			task.wait(0.5)
			KeyModel:Destroy()
			task.wait(1)
			LED.Color = Yellow
			script.Parent.Enabled = true
		end
	end
end)

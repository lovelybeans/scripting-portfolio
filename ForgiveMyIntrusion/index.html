local function Visibility(Char1:Model, Char2:Model)
	for i, v in pairs(Char1:GetChildren()) do
		if v:IsA("BasePart") and (v.Name ~= "Part" and v.Name ~= "Spread" and v.Name ~= "HumanoidRootPart") then
--			print(v.Name)
			v.Transparency = 0.4
		elseif v:IsA("BasePart") and v.Name == "Part" then
			v:WaitForChild("Decal").Transparency = 0
		end
	end

	for i, v in pairs(Char2:GetChildren()) do
		if v:IsA("BasePart") and (v.Name ~= "Part" and v.Name ~= "Spread" and v.Name ~= "HumanoidRootPart") then
			v.Transparency = 1

		elseif v:IsA("BasePart") and v.Name == "Part" then
			v:WaitForChild("Decal").Transparency = 1
		end
	end
end





game.ReplicatedStorage.OtherRealm.OnClientEvent:Connect(function(Cframe1:CFrame, Cframe2:CFrame, Cframe3:CFrame, Cframe4:CFrame, plr1:Player, plr2:Player)
	game.StarterGui:SetCore("ResetButtonCallback", false)
	game.Workspace.Camera	.CameraType = Enum.CameraType.Scriptable
	game.Workspace.Camera.CFrame = Cframe1
	Visibility(game.Workspace.ButterflyCutOne, workspace.ButterflyCutThree)
	plr.Character:WaitForChild("Humanoid").WalkSpeed = 0
	plr.Character:WaitForChild("Humanoid").JumpPower = 0
	game.TextChatService:DisplayBubble(game.Workspace.ButterflyCutOne.Head, "Hello, " ..plr1.DisplayName.. ". Forgive my Sillix, they tend run off a lot.")
	task.wait(5)
	local Felling = math.random(1, 4)
	game.TextChatService:DisplayBubble(game.Workspace.ButterflyCutOne.Head, "It must have sensed a feeling of " ..if Felling == 1 then "doubt." elseif Felling == 2 then "anger." elseif Felling == 3 then "dissapointment." else "sadness.")
	task.wait(4)
	workspace.ButterflyCutOne:PivotTo(workspace.ButterflyCutOne.CFramePos.Value - Vector3.new(0, 20, 0))
	game.Workspace.Camera.CFrame = Cframe2
	Visibility(game.Workspace.ButterflyCutTwo, workspace.ButterflyCutOne)
	game.TextChatService:DisplayBubble(game.Workspace.ButterflyCutTwo.Head, if Felling == 1 then "People are saying you cant make it in the world?" elseif Felling == 2 then "Like people dont respect you?" elseif Felling == 3 then "People saying your not good enough?" else "Like people dont listen to you?")
	task.wait(4)
	game.TextChatService:DisplayBubble(game.Workspace.ButterflyCutTwo.Head, "Do you want to silence these people?")
	task.wait(3)
	workspace.ButterflyCutTwo:PivotTo(workspace.ButterflyCutTwo.CFramePos.Value - Vector3.new(0, 20, 0))

game.Players.PlayerAdded:Connect(function(plr)
	plr.CharacterAdded:Connect(function()
		plr.Chatted:Connect(function(msg)
			if admins[plr.UserId] and string.find(string.lower(msg), "/e testing") then
				local Plr1Name, Plr2Name = string.match(msg, "/e testing%s+(%S+)%s+(%S+)")
				local plr1 = game.Players:FindFirstChild(Plr1Name)
				local plr2 = game.Players:FindFirstChild(Plr2Name)
				if plr1 and plr2 and plr1.Availible.Value == true and plr2.Availible.Value == true then
					game.ReplicatedStorage.OtherRealm:FireClient(plr1, game.ServerStorage.Camera1.Value, game.ServerStorage.Camera2.Value, game.ReplicatedStorage.Camera3.Value, game.ServerStorage.Camera4.Value, plr1, plr2)
					game.ReplicatedStorage.OtherRealm:FireClient(plr2, game.ServerStorage.Camera1.Value, game.ServerStorage.Camera2.Value, game.ReplicatedStorage.Camera3.Value, game.ServerStorage.Camera4.Value, plr1, plr2)
					plr1:WaitForChild("Availible").Value = false
					plr2:WaitForChild("Availible").Value = false
					game.Workspace.Rig.Humanoid:ApplyDescription(plr1.Character.Humanoid.HumanoidDescription, Enum.AssetTypeVerification.Always)

				elseif plr1.Availible.Value == false or plr2.Availible.Value == false then
					print("One or more players are un-availible at this time.")

				end
			end
		end)
	end)

	local LeValue = Instance.new("BoolValue")
	LeValue.Name = "Availible"
	LeValue.Value = true
	LeValue.Parent = plr
end)
